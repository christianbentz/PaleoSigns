Load libraries

If the libraries are not installed yet, you need to install them using, for example, the command: install.packages(“ggplot2”).

library(ggplot2)
library(ggExtra)
library(ggpubr)
library(gridExtra)

Load Data

Load csv file with quantitative feature values for sign sequences per object.

# load signbase estimations for strings on objects
sign.data <- read.csv("~/Github/UpperPalCombinatorics/results/signBase_features.csv")
# select Aurignacian objects
sign.data <- sign.data[sign.data$archaeological_unit == "Aurignacien", ]

PLOT

Entropy rate vs. Unigram entropy

Entropy rate vs. unigram entropy for the original sequences.

huni.hrate.orig.plot <- ggplot(sign.data, aes(x = huni.chars, y =  hrate.chars, 
                                    color = site_name, size = num.char)) + 
  geom_point(alpha = 0.3) +
  #geom_density2d(alpha = .8) +
  labs(x = "Unigram entropy for characters", y = "Entropy rate for characters")
huni.hrate.orig.plot <- ggMarginal(huni.hrate.orig.plot, groupFill = T, groupColour = T, 
                                  type = "density")
#huni.hrate.chars.plot

Entropy rate vs. unigram entropy for the original sequences.

huni.hrate.random.plot <- ggplot(sign.data, aes(x = huni.chars, y =  hrate.chars.random,

Combine figures.

combined.plot <- grid.arrange(huni.hrate.orig.plot, huni.hrate.random.plot, ncol = 2)

Safe complete figure to file

ggsave("~/Github/UpperPalCombinatorics/figures/combinedPlot_signBase_random.pdf", 
       combined.plot, dpi = 300, width = 12, height = 6, device = cairo_pdf)
