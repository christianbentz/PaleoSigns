Proto-Cuneiform Preprocessing

Description

This file enables preprocessing of cuneiform tablet transliterations, these can be downloaded from CDLI (https://cdli.mpiwg-berlin.mpg.de/search). The extraction of proto-cuneiform sign codings is done with the sign lists provided by CDLI. In their syntax, repetitions of signs are indicated by a number followed by the sign type in parentheses, e.g. 6(N34) 1(N45) 3(N14) 5(N01), SZE~a (see Figure below). This is here expanded to yield individual sign tokens, e.g. N34 N34 … N34 N45 N14 N14 N14, etc. We also include signs which are followed by ‘#’ (broken but identifiable signs), but we exclude signs followed by ‘?’ (questionable). We here only consider the periods “Uruk V”, “Uruk IV”, and “Uruk III”, as these are typically seen as the proto-cuneiform periods, i.e. before genuine writing of spoken Sumerian developed.

Get vector of IDs for tablets.

# the ampersand followed by P identifies a single tablet transliteration
tablet.id <- textlines[grep(pattern = "\\&P", textlines)]
# check length

# remove ampersand

Extract and concatenate the transliterations.

# get chunks of text by ampersand plus "P" (at the beginning of each tablet)
textchunks <- str_split(textstring, pattern = "\\&P")
# unlist and remove first element (empty)
textchunks <- unlist(textchunks)[-1]
# remove the header
textchunks <- gsub("\\=.*qpc", "", textchunks)

# use signlist provided by CDLI to extract only signs
# create a duplicate signlist with `#' following after the signs,
# this means the sign is partly broken but identifiable
signlist.hashtag <- paste0(signlist,'#')
# concatenate the two lists (i.e. vectors)
signlist <- c(signlist, signlist.hashtag)
# first some characters in the sign list need to be escaped
signlist <- escapeRegex(signlist)
# add white spaces after character strings in signlist
# this helps with only extracting full string matches
signlist <- paste0(signlist,' ')
# remove 'X' as this is a placeholder for unidentified signs
signlist <- signlist[!signlist == 'X ']

# run for loop to extract signs based on sign list
# initialize vectors
signs.orig.vec <- c()
signs.expand.vec <- c()
for (i in 1:length(textchunks)){
  # note that the order of signs is here sometimes changed, 
  # as it follows the order of signs in sign list
  signs.orig <- unlist(str_extract_all(textchunks[i], paste(signlist, collapse = "|")))
  signs <- unlist(str_extract_all(textchunks[i], paste(signlist, collapse = "|")))
  # trim white spaces around signs
  signs.orig <- trimws(signs.orig)
  signs <- trimws(signs)
  # remove the hashtag
  signs <- gsub("#", "", signs)
  # unravel the multiplication syntax to get multiple sign tokens
  try(silent = TRUE, { # if the processing fails for a certain row, there will be no output for this row,
  # but the try() function allows the loop to keep running
  for (j in 1:length(signs)){
    sign.syntax <- signs[j]
    # get only number
    remainder <- sub(pattern = "\\(.*\\)", "", sign.syntax)
      # if number then multiply with sign type, else 
      # leave sign syntax as is
      if (!grepl("[^0-9]", remainder)){
        # get sign.type inside round brackets
        sign.type <- gsub(".*\\((.*)\\).*", "\\1", sign.syntax)
        # repeat given the number ("remainder" from above)
        sign.tokens <- rep(sign.type, remainder)
        # collaps into character string
        sign.tokens <- paste(sign.tokens, collapse = " ")
      } else {
        sign.tokens <- sign.syntax
      }
    signs[j] <- sign.tokens
  }}) # end of try() function
  # collapse the signs into one string
  signs <- paste(signs, collapse = " _ ")
  signs.orig <- paste(signs.orig, collapse = " _ ")
  signs.orig.vec <- c(signs.orig.vec, signs.orig)
  signs.expand.vec <- c(signs.expand.vec, signs)
}
# compile final data frame
cunei.data <- data.frame(tablet.id, signs.orig.vec, signs.expand.vec)

Remove rows with empty sign codings

cunei.data <- cunei.data[-which(cunei.data$signs.orig.vec == ""), ]

Select those with at least a given number of cuneiform signs.

cunei.data <- cunei.data[lengths(strsplit(cunei.data$signs.expand.vec, " ")) > 1, ]
